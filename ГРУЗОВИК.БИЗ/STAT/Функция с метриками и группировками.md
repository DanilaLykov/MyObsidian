`from fastapi import APIRouter, Query`
`from sqlalchemy import select, func, case, text, desc, asc, literal`
`from app.models.metric import Metric, TYPE_MAP`
`from app.db import async_session`
`from typing import List, Optional`

`router = APIRouter()`

# `Конфиг составных метрик`
`COMPOSITE_METRICS = {`
    `"targetActions": ["kpRequest", "callRequest", "priceOffer", "chatMessage"],`
    `# Можно добавить другие составные метрики`
`}`

# `Поддерживаемые операторы фильтров`
`OPERATORS = {`
    `"==": lambda col, val: col == val,`
    `"!=": lambda col, val: col != val,`
    `">": lambda col, val: col > val,`
    `"<": lambda col, val: col < val,`
    `">=": lambda col, val: col >= val,`
    `"<=": lambda col, val: col <= val,`
    `"in": lambda col, val: col.in_(val.split(",")),`
`}`

`@router.get("/metrics/report")`
`async def metrics_report(`
    `metrics: List[str] = Query(["views"]),`
    `dimensions: List[str] = Query([]),`
    `filters: List[str] = Query([]),`
    `date1: Optional[int] = None,`
    `date2: Optional[int] = None,`
    `group_by_date: Optional[str] = Query(None, description="day, week, month"),`
    `sort: Optional[str] = None,`
    `limit: int = 10,`
`):`
    `async with async_session() as session:`
        `# 1️⃣ Базовый SELECT с группировкой по dimensions`
        `group_by_cols = [getattr(Metric, d) for d in dimensions] if dimensions else []`

        `# Агрегирование по дате (day/week/month)`
        `if group_by_date:`
            `if group_by_date == "day":`
                `date_col = func.to_char(func.to_timestamp(Metric.time), "YYYY-MM-DD").label("day")`
            `elif group_by_date == "week":`
                `date_col = func.to_char(func.to_timestamp(Metric.time), "IYYY-IW").label("week")`
            `elif group_by_date == "month":`
                `date_col = func.to_char(func.to_timestamp(Metric.time), "YYYY-MM").label("month")`
            `else:`
                `date_col = literal(None)`
            `if date_col is not None:`
                `group_by_cols.append(date_col)`

        `q = select(*group_by_cols).group_by(*group_by_cols) if group_by_cols else select()`

        `# 2️⃣ Добавляем обычные и составные метрики`
        `for m in metrics:`
            `if m in TYPE_MAP.values():`
                `# обычная метрика`
                `db_type = next((k for k, v in TYPE_MAP.items() if v == m), None)`
                `if db_type:`
                    `q = q.add_columns(`
                        `func.sum(case((Metric.type == db_type, 1), else_=0)).label(m)`
                    `)`
            `elif m in COMPOSITE_METRICS:`
                `# составная метрика`
                `q = q.add_columns(`
                    `func.sum(case((Metric.type.in_(COMPOSITE_METRICS[m]), 1), else_=0)).label(m)`
                `)`
            `else:`
                `# если метрика неизвестная, возвращаем 0`
                `q = q.add_columns(literal(0).label(m))`

        `# 3️⃣ Применяем фильтры`
        `for f in filters:`
            `for op_key, op_func in OPERATORS.items():`
                `if op_key in f:`
                    `field, val = f.split(op_key)`
                    `field = field.strip()`
                    `val = val.strip()`
                    `q = q.where(op_func(getattr(Metric, field), val))`
                    `break`

        `# 4️⃣ Фильтр по дате`
        `if date1 and date2:`
            `q = q.where(Metric.time.between(date1, date2))`

        `# 5️⃣ Сортировка`
        `if sort:`
            `direction = desc if sort.startswith("-") else asc`
            `field = sort.lstrip("-")`
            `q = q.order_by(direction(text(field)))`

        `# 6️⃣ Лимит`
        `q = q.limit(limit)`

        `# 7️⃣ Выполнение запроса`
        `result = await session.execute(q)`
        `return result.mappings().all()`



**ДЛЯ KPI**

GET /metrics/report?
metrics=shows,views,targetActions,calls,chatMessages,phoneViews,kpRequests&
date1=20250801&
date2=20250830&
sort=-targetActions&
limit=10

**Разбор параметров:**

- `metrics=...` — список метрик, которые нужно посчитать.
- `date1` / `date2` — период.
- `sort=-targetActions` — сортировка по убыванию целевых действий.
- `limit=10` — вернуть максимум 10 записей.
- `dimensions` не указан → агрегат по всему набору, без разбивки.
  
  Пример JSON-ответа
  
  [
  {
    "shows": 500,
    "views": 200,
    "targetActions": 15,
    "calls": 3,
    "chatMessages": 5,
    "phoneViews": 7,
    "kpRequests": 2
  }
]
